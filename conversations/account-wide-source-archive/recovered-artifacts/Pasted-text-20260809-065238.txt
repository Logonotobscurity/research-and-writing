import { createOpenAICompatible } from "@ai-sdk/openai-compatible";
import { z } from "zod";

/**
 * OpenRouter AI Gateway provider. Server-only: the API key never reaches the client.
 * 
 * Supports:
 * - OpenAI-compatible /api/v1/chat/completions endpoint
 * - Structured JSON output via schema parameter
 * - Tool calling
 * - Streaming
 * - Usage tracking
 * - Custom HTTP headers for rate limiting and tracking
 */
export function createOpenRouterProvider(apiKey: string) {
  if (!apiKey) {
    throw new Error("OPENROUTER_API_KEY is not set");
  }

  const siteUrl = process.env.OPENROUTER_SITE_URL || "http://localhost:3000";
  const siteName = process.env.OPENROUTER_SITE_NAME || "CLIPIN";

  return createOpenAICompatible({
    name: "openrouter",
    baseURL: "https://openrouter.ai/api/v1",
    apiKey,
    headers: {
      "HTTP-Referer": siteUrl,
      "X-Title": siteName,
    },
    defaultQuery: {
      include_reasoning: "true",
    },
  });
}

 * Get configured model from environment or use default.

export function getConfiguredModel(): string {
  return (

/**
 * Build a complete prompt from template and variables
 */
export function buildPrompt(
  config: PromptConfig,
  variables: Record<string, any>
): string {
  let prompt = config.userPromptTemplate;

  // Replace all {variable} with actual values
  Object.entries(variables).forEach(([key, value]) => {
    const regex = new RegExp(`{${key}}`, "g");
    prompt = prompt.replace(regex, String(value || ""));
  });

  return prompt;
}

import { generateObject } from "ai";
import { z } from "zod";
import {
  getOpenRouterProvider,
  getConfiguredModel,
  LLMResponse,
  DEFAULT_LLM_CONFIG,
  LLMRequestConfig,
} from "./ai-gateway.server";
import {
  getPromptConfig,
  buildPrompt,
  AITaskId,
  PromptConfig,
} from "./prompt-registry";

/**
 * CLIPIN AI Task Executor
 *
 * Orchestrates LLM calls with:
 * - Prompt registry lookup
 * - Input validation
 * - Structured output parsing
 * - Error handling with fallbacks
 * - Usage tracking
 */

export interface AITaskInput {
  taskId: AITaskId;
  variables: Record<string, any>;
  config?: Partial<LLMRequestConfig>;
}

export interface AITaskResult<T = any> {
  success: boolean;
  data?: T;
  error?: string;
  usage?: {
    inputTokens: number;
    outputTokens: number;
  };
  model: string;
  timestamp: Date;
}

/**
 * Execute an AI task from the prompt registry
 */
export async function executeAITask<T = any>(
  input: AITaskInput
): Promise<AITaskResult<T>> {
  const startTime = Date.now();

  try {
    // 1. Validate task ID and fetch prompt config
    const promptConfig = getPromptConfig(input.taskId);
    if (!promptConfig) {
      return {
        success: false,
        error: `Unknown task: ${input.taskId}`,
        model: getConfiguredModel(),
        timestamp: new Date(),
      };
    }

    // 2. Build the complete user prompt
    const userPrompt = buildPrompt(promptConfig, input.variables);

    // 3. Merge configuration
    const config = { ...DEFAULT_LLM_CONFIG, ...input.config };

    // 4. Get OpenRouter provider
    const provider = getOpenRouterProvider();
    const model = getConfiguredModel();

    // 5. Execute LLM call with structured output
    const result = await generateObject({
      model: provider(model),
      system: promptConfig.systemInstructions,
      prompt: userPrompt,
      schema: promptConfig.outputSchema,
      temperature: promptConfig.modelConfig.temperature,
      topP: promptConfig.modelConfig.topP,
      maxTokens: config.maxTokens,
    });

    return {
      success: true,
      data: result.object as T,
      usage: {
        inputTokens: result.usage?.promptTokens || 0,
        outputTokens: result.usage?.completionTokens || 0,
      },
      model,
      timestamp: new Date(),
    };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : String(error);

    // Check for fallback content
    const promptConfig = getPromptConfig(input.taskId);
    if (promptConfig?.fallback?.enabled) {
      return {
        success: true,
        data: { fallback: promptConfig.fallback.content } as T,
        error: `Used fallback: ${errorMessage}`,
        model: getConfiguredModel(),
        timestamp: new Date(),
      };
    }

    return {
      success: false,
      error: errorMessage,
      model: getConfiguredModel(),
      timestamp: new Date(),
    };
  }
}

/**
 * Batch execute multiple AI tasks
 */
export async function batchExecuteAITasks(
  inputs: AITaskInput[]
): Promise<AITaskResult[]> {
  return Promise.all(inputs.map((input) => executeAITask(input)));
}

/**
 * Execute with automatic retry logic
 */
export async function executeAITaskWithRetry<T = any>(
  input: AITaskInput,
  maxRetries: number = 2
): Promise<AITaskResult<T>> {
  let lastError: AITaskResult<T> | null = null;

  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const result = await executeAITask<T>(input);

    if (result.success) {
      return result;
    }

    lastError = result;

    // Exponential backoff
    if (attempt < maxRetries) {
      const delayMs = Math.pow(2, attempt) * 1000;
      await new Promise((resolve) => setTimeout(resolve, delayMs));
    }
  }

  return lastError || {
    success: false,
    error: "Unknown error after retries",
    model: getConfiguredModel(),
    timestamp: new Date(),
  };
}

/**
 * Stream AI task result (for real-time updates)
 * Note: streaming is handled by the Vercel AI SDK
 */
export async function streamAITask(
  input: AITaskInput,
  onChunk?: (chunk: string) => void
) {
  const provider = getOpenRouterProvider();
  const model = getConfiguredModel();
  const promptConfig = getPromptConfig(input.taskId);

  if (!promptConfig) {
    throw new Error(`Unknown task: ${input.taskId}`);
  }

  const userPrompt = buildPrompt(promptConfig, input.variables);

  // For streaming, use generateText instead of generateObject
  // This is handled by the caller depending on UI requirements
  return {
    provider,
    model,
    system: promptConfig.systemInstructions,
    prompt: userPrompt,
  };
}

/**
 * Validate AI task input before execution
 */
export function validateAITaskInput(input: AITaskInput): {
  valid: boolean;
  errors: string[];
} {
  const errors: string[] = [];

  if (!input.taskId) {
    errors.push("taskId is required");
  }

  if (!input.variables || typeof input.variables !== "object") {
    errors.push("variables must be an object");
  }

  return {
    valid: errors.length === 0,
    errors,
  };
}

/**
 * Validate URL is safe to crawl
 */
export function validateCrawlUrl(
  url: string,
  config: CrawlConfig = DEFAULT_CRAWL_CONFIG
): {
  valid: boolean;
  error?: string;
} {
  try {
    const parsed = new URL(url);
    const hostname = parsed.hostname || "";

    // Check protocol
    if (!["http", "https"].includes(parsed.protocol.replace(":", ""))) {
      return { valid: false, error: "Only HTTP(S) URLs allowed" };
    }

    // Check against allowed domains
    if (config.allowedDomains && config.allowedDomains.length > 0) {
      const isAllowed = config.allowedDomains.some((domain) =>
        hostname.endsWith(domain)
      );
      if (!isAllowed) {
        return { valid: false, error: `Domain not in allowlist: ${hostname}` };
      }
    }

    // Check against blocked domains
    if (config.blockedDomains && config.blockedDomains.length > 0) {
      const isBlocked = config.blockedDomains.some((domain) =>
        hostname.endsWith(domain)
      );
      if (isBlocked) {
        return { valid: false, error: `Domain is blocked: ${hostname}` };
      }
    }

    // Prevent SSRF: block private IP ranges
    const privatePatterns = [
      /^localhost$/,
      /^127\./,
      /^192\.168\./,
      /^10\./,
      /^172\.(1[6-9]|2[0-9]|3[01])\./,
      /^::1$/,
      /^fc00:/,
    ];

    if (privatePatterns.some((pattern) => pattern.test(hostname))) {
      return {
        valid: false,
        error: "Private IP addresses not allowed (SSRF protection)",
      };
    }

    return { valid: true };
  } catch (error) {
    return { valid: false, error: `Invalid URL: ${error}` };
  }
}

/**
 * Fetch and extract content from a single URL
 *
 * This is a placeholder for server-side implementation.
 * In production, you'd use a headless browser or HTML parsing library.
 */
export async function extractContent(
  url: string,
  config: CrawlConfig = DEFAULT_CRAWL_CONFIG
): Promise<ExtractedContent | null> {
  // Validate URL
  const validation = validateCrawlUrl(url, config);
  if (!validation.valid) {
    throw new Error(validation.error || "Invalid URL");
  }

  try {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), config.timeoutMs);

    const response = await fetch(url, {
      method: "GET",
      headers: {
        "User-Agent": config.userAgent,
        Accept: "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
      },
      signal: controller.signal,
    });

    clearTimeout(timeoutId);

    // Check content type
    const contentType = response.headers.get("content-type") || "";
    if (!contentType.includes("text/html")) {
      throw new Error(`Unsupported content type: ${contentType}`);
    }

    // Check size
    const contentLength = parseInt(
      response.headers.get("content-length") || "0",
      10
    );
    if (contentLength > config.maxBytesPerPage) {
      throw new Error(
        `Content exceeds max size: ${contentLength} > ${config.maxBytesPerPage}`
      );
    }

    // Check for robots.txt directive
    const robotsHeader = response.headers.get("x-robots-tag");
    if (config.respectRobots && robotsHeader) {
      if (robotsHeader.includes("noindex") || robotsHeader.includes("none")) {
        throw new Error("URL has noindex directive");
      }
    }

    const html = await response.text();

    // Extract metadata (simplified)
    const titleMatch = html.match(/<title>(.+?)<\/title>/i);
    const descMatch = html.match(
      /<meta\s+name="description"\s+content="(.+?)"/i
    );
    const canonicalMatch = html.match(
      /<link\s+rel="canonical"\s+href="(.+?)"/i
    );

    // Strip HTML (simplified - in production use a library like jsdom)
    const cleanContent = html
      .replace(/<script[^>]*>.*?<\/script>/gi, "")
      .replace(/<style[^>]*>.*?<\/style>/gi, "")
      .replace(/<[^>]+>/g, " ")
      .replace(/\s+/g, " ")
      .trim();

    return {
      url,
      canonicalUrl: canonicalMatch ? canonicalMatch[1] : url,
      title: titleMatch ? titleMatch[1] : undefined,
      description: descMatch ? descMatch[1] : undefined,
      content: cleanContent,
      contentType,
      metadata: {
        statusCode: response.status,
        headers: Object.fromEntries(response.headers.entries()),
      },
    };
  } catch (error) {
    throw error;
  }
}

/**
 * Crawl a URL and extract content (with depth limit)
 *
 * Note: This is a simplified implementation.
 * For production, use a proper crawling library with queue management.
 */
export async function crawlUrl(
  rootUrl: string,
  config: CrawlConfig = DEFAULT_CRAWL_CONFIG
): Promise<CrawlResult> {
  const startTime = Date.now();
  const result: CrawlResult = {
    rootUrl,
    pages: [],
    totalBytesProcessed: 0,
    pagesAttempted: 0,
    pagesSucceeded: 0,
    pagesFailed: 0,
    errors: [],
    crawlDuration: 0,
    timestamp: new Date(),
  };

  // Validate root URL
  const validation = validateCrawlUrl(rootUrl, config);
  if (!validation.valid) {
    result.errors.push({
      url: rootUrl,
      error: validation.error || "Invalid URL",
    });
    result.crawlDuration = Date.now() - startTime;
    return result;
  }

  try {
    const content = await extractContent(rootUrl, config);
    if (content) {
      result.pages.push(content);
      result.totalBytesProcessed += content.content.length;
      result.pagesSucceeded += 1;
    }
  } catch (error) {
    result.errors.push({
      url: rootUrl,
      error: error instanceof Error ? error.message : String(error),
    });
    result.pagesFailed += 1;
  }

  result.pagesAttempted = 1;
  result.crawlDuration = Date.now() - startTime;
  return result;
}

/**
 * Chunk extracted content for LLM consumption
 */
export function chunkContent(
  content: ExtractedContent,
  chunkSize: number = 2000,
  overlap: number = 200
): string[] {
  const chunks: string[] = [];
  const text = content.content;

  for (let i = 0; i < text.length; i += chunkSize - overlap) {
    chunks.push(text.slice(i, i + chunkSize));
  }

  return chunks;
}

/**
 * Deduplicate content across multiple pages
 */
export function deduplicateContent(
  pages: ExtractedContent[]
): ExtractedContent[] {
  const seen = new Set<string>();
  const unique: ExtractedContent[] = [];

  for (const page of pages) {
    const contentHash = hashString(page.content);
    if (!seen.has(contentHash)) {
      seen.add(contentHash);
      unique.push(page);
    }
  }

  return unique;
}

/**
 * Simple hash function for deduplication
 */
function hashString(str: string): string {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i);
    hash = (hash << 5) - hash + char;
    hash = hash & hash; // Convert to 32bit integer
  }
  return hash.toString(36);
}

/**
 * Format crawl result for LLM context
 */
export function formatCrawlResultForLLM(result: CrawlResult): string {
  let formatted = `# Research Results\n\n`;
  formatted += `**Source:** ${result.rootUrl}\n`;
  formatted += `**Pages Processed:** ${result.pagesSucceeded}/${result.pagesAttempted}\n\n`;

  for (const page of result.pages) {
    formatted += `## ${page.title || "Untitled"}\n`;
    formatted += `**URL:** ${page.canonicalUrl || page.url}\n`;
    if (page.description) {
      formatted += `**Description:** ${page.description}\n`;
    }
    formatted += `\n${page.content.slice(0, 1000)}...\n\n`;
  }

  if (result.errors.length > 0) {
    formatted += `## Errors\n`;
    for (const error of result.errors) {
      formatted += `- ${error.url}: ${error.error}\n`;
    }
  }

  return formatted;
}

3. Use in a Server Action
Create app/actions/ideas.ts:

TypeScript
"use server"

import { generateVideoIdeas } from "@/lib/ai";

export async function generateIdeasAction(topic: string) {
  const result = await generateVideoIdeas({
    topic,
    platform: "tiktok",
    audience: "content creators",
  });

  if (!result.success) {
    throw new Error(`Failed to generate ideas: ${result.error}`);
  }

  return result.data?.ideas || [];
}
Create app/components/IdeaForm.tsx:

TypeScript
"use client"

import { generateIdeasAction } from "@/app/actions/ideas";
import { useState } from "react";

export function IdeaForm() {
  const [ideas, setIdeas] = useState([]);
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setLoading(true);

    try {
      const formData = new FormData(e.currentTarget);
      const topic = formData.get("topic") as string;
      const result = await generateIdeasAction(topic);
      setIdeas(result);
    } catch (error) {
      console.error("Error:", error);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div>
      <form onSubmit={handleSubmit}>
        <input name="topic" placeholder="Enter topic..." required />
        <button type="submit" disabled={loading}>
          {loading ? "Generating..." : "Generate Ideas"}
        </button>
      </form>

      <ul>
        {ideas?.map((idea, i) => (
          <li key={i}>
            <h3>{idea.title}</h3>
            <p>{idea.concept}</p>
          </li>
        ))}
      </ul>
    </div>
  );
}
